## Task 0：SAI1 + DMA + 时钟 + PA_EN 引脚（CubeMX 风格手写配置）

本工程不从 CubeMX 重生成（见记忆 [[eide_builder_params]]），所以 SAI 初始化按现有 `i2c.c`/`spi.c` 的手写风格新增。

**目标时钟（关键算式）：**
- MCLK = **12.288 MHz** = 256×48000 = 768×16000 = 1536×8000 → **一个 MCLK 覆盖三档采样率**。
- SAI1 内核时钟取 12.288 MHz 的整数倍（如 49.152 MHz，由 PLL2 或 PLL3 P/Q 产生），再由 SAI 的 MCKDIV 输出 12.288 MHz MCLK。
- 帧：I2S Philips 标准，16-bit 数据，每帧 2 slot（立体声帧时序），ES8311 单声道数据放左声道；BCLK = 32×Fs。

**Files:**

- [ ] **Step 1: 确认 PA_EN 与 SAI1 引脚的实际 MCU 管脚**

引脚已从原理图 + 封装脚位交叉核对确认（LED=PB12=脚33、POWER_CTL=PB7=脚59、DEBUG=PA9/PA10=脚42/43 验证封装编号）：

| 功能 | MCU GPIO | 封装脚 | ES8311 引脚 |
|------|----------|--------|------------|
| I2C2_SCL | PB13 | 34 | CCLK(1) |
| I2C2_SDA | PB14 | 35 | CDATA(19) |
| SAI1_MCLK_A | PB8 | 61 | MCLK(2) |
| SAI1_SCK_A (BCLK) | PA8 | 41 | SCLK(6) |
| SAI1_FS_A (LRCK) | PB9 | 62 | LRCK(8) |
| SAI1_SD_A (数据，codec→MCU) | PC3 | 11 | ASDOUT(7) |
| PA_EN (电源门控) | **PB15** | 36 | →AO3401A PFET→ PAVCC |

- I2C2(PB13/PB14)与 PCF85063 RTC 共用，ES8311 地址 0x18，RTC 0x51；上拉 R27/R28 4.7K 已在板上。
- **PA_EN 低电平有效**：AO3401A 为 P 沟道，源=3V3、漏=PAVCC、栅经 R32(10K) 上拉到 3V3。默认关断（上电安全）；PA_EN 拉低 → 导通给 codec 供电。
- SAI1 AF：PB8/PB9 走 AF3，PA8/PC3 走 AF13（U5 的 SAI1 同时在 AF3/AF13），实现时用 CubeMX 或数据手册复用表确认。

- [ ] **Step 5: 新建 sai.c —— 时钟、GPIO、SAI、DMA 配置**

`Core/Src/sai.c`。要点：① 用 `HAL_RCCEx_PeriphCLKConfig` 把 SAI1 内核时钟设为 12.288MHz×N；② 配置 4 个 SAI1 复用引脚；③ SAI 主收 I2S 16-bit、MCLK 输出开；④ GPDMA 通道循环模式、半字。下面给出骨架（PLL 具体寄存器值与 GPDMA 通道号实现时据本工程时钟树/可用通道确定，并在 Step 8 用示波器验证 MCLK=12.288MHz）：

```c
#include "sai.h"

SAI_HandleTypeDef hsai_BlockA1;
DMA_HandleTypeDef handle_GPDMA1_Channel_SAI;

/* 由 sample_rate 反推 SAI AudioFrequency 枚举 */
static uint32_t sai_audio_freq(uint32_t sr)
{
  switch (sr) {
    case 8000:  return SAI_AUDIO_FREQUENCY_8K;
    case 16000: return SAI_AUDIO_FREQUENCY_16K;
    case 48000: return SAI_AUDIO_FREQUENCY_48K;
    default:    return SAI_AUDIO_FREQUENCY_16K;
  }
}

HAL_StatusTypeDef MX_SAI1_Init(uint32_t sample_rate_hz)
{
  /* 1) SAI1 内核时钟 = 12.288MHz 的整数倍（PLL 提供）。
   *    实现时在 main.c SystemClock_Config 之后/此处用 RCCEx 配好 PLL2 或 PLL3
   *    输出 49.152MHz 给 SAI1，使 MCKDIV 能精确得到 256×Fs。 */
  RCC_PeriphCLKInitTypeDef pclk = {0};
  pclk.PeriphClockSelection = RCC_PERIPHCLK_SAI1;
  pclk.Sai1ClockSelection   = RCC_SAI1CLKSOURCE_PLL2;  /* 据时钟树确认 */
  /* pclk.PLL2.* = 产生 49.152MHz 的参数（实现时计算填写） */
  if (HAL_RCCEx_PeriphCLKConfig(&pclk) != HAL_OK) return HAL_ERROR;

  /* 2) GPIO 复用（MCLK/SCK/FS/SD，AF 据数据手册 alternate function 表填写） */
  __HAL_RCC_SAI1_CLK_ENABLE();
  __HAL_RCC_GPIOA_CLK_ENABLE();   /* 据实际脚位使能对应端口 */
  __HAL_RCC_GPIOB_CLK_ENABLE();
  /* GPIO_InitStruct: Mode=AF_PP, Pull=NOPULL, Speed=VERY_HIGH, Alternate=GPIO_AFx_SAI1
     对 SAI1_MCLK_A / SAI1_SCK_A / SAI1_FS_A / SAI1_SD_A 四脚分别 HAL_GPIO_Init */

  /* 3) SAI 主收、I2S Philips、16-bit、MCLK 输出 */
  hsai_BlockA1.Instance            = SAI1_Block_A;
  hsai_BlockA1.Init.Protocol       = SAI_FREE_PROTOCOL;
  hsai_BlockA1.Init.AudioMode      = SAI_MODEMASTER_RX;
  hsai_BlockA1.Init.DataSize       = SAI_DATASIZE_16;
  hsai_BlockA1.Init.FirstBit       = SAI_FIRSTBIT_MSB;
  hsai_BlockA1.Init.ClockStrobing  = SAI_CLOCKSTROBING_FALLINGEDGE; /* ADC 在 SCLK 下降沿出数据（数据手册 5.x） */
  hsai_BlockA1.Init.Synchro        = SAI_ASYNCHRONOUS;
  hsai_BlockA1.Init.MckOutput      = SAI_MCK_OUTPUT_ENABLE;
  hsai_BlockA1.Init.AudioFrequency = sai_audio_freq(sample_rate_hz);
  hsai_BlockA1.Init.MckOverSampling= SAI_MCK_OVERSAMPLING_DISABLE;
  /* I2S 标准帧：32-bit 帧、16-bit slot×2，slot0 取数据。用 HAL_SAI_InitProtocol
     或手填 FrameInit/SlotInit。推荐： */
  if (HAL_SAI_InitProtocol(&hsai_BlockA1, SAI_I2S_STANDARD, SAI_PROTOCOL_DATASIZE_16BIT, 2) != HAL_OK)
    return HAL_ERROR;

  return HAL_OK;
}

void MX_SAI1_DeInit(void)
{
  HAL_SAI_DeInit(&hsai_BlockA1);
}
```

## Task 1：ES8311 codec 驱动（I2C 初始化序列）

ES8311 I2C 7-bit 地址 **0x18**（CE 接地，数据手册 §I2C：芯片地址 `0011 00x`）。HAL 用 8-bit 地址：写 `0x30`、读 `0x31`。寄存器初始化采用业界通用的 ES8311 **纯 ADC（mic 输入）** 序列（ESP-ADF `es8311.c` 参考实现），并在 Task 7 台架用已知音验证/微调增益。

- [ ] **Step 2: 新建 es8311.c —— I2C 读写 + 互斥锁**

I2C2 与 PCF85063 RTC 共用，加互斥锁（在 `app_freertos.c` 已有 RTOS，定义一个全局 `osMutexId_t g_i2c2_mutex`，本步骤先用 `HAL_I2C_Mem_Write/Read` 直接读写，互斥锁在 Task 7 接线时统一加在调用处）。

```c
#include "es8311.h"
#include "i2c.h"
#include "cmsis_os2.h"

static int es_write(uint8_t reg, uint8_t val)
{
  return (HAL_I2C_Mem_Write(&hi2c2, ES8311_I2C_ADDR_8BIT, reg,
            I2C_MEMADD_SIZE_8BIT, &val, 1, 100) == HAL_OK) ? 0 : -1;
}
static int es_read(uint8_t reg, uint8_t *val)
{
  return (HAL_I2C_Mem_Read(&hi2c2, ES8311_I2C_ADDR_8BIT, reg,
            I2C_MEMADD_SIZE_8BIT, val, 1, 100) == HAL_OK) ? 0 : -1;
}

int ES8311_Probe(void)
{
  uint8_t id1 = 0, id2 = 0;
  if (es_read(0xFD, &id1) != 0) return -1;   /* ES8311 CHIP_ID1 ≈ 0x83 */
  if (es_read(0xFE, &id2) != 0) return -1;   /* CHIP_ID2 ≈ 0x11 */
  return (id1 == 0x83 && id2 == 0x11) ? 0 : -2;
}
```

- [ ] **Step 3: es8311.c —— 时钟系数（MCLK/Fs 恒为 256）**

**重要修正（Task 0 发现）**：SAI 主机的 MCLK 锁定为 256×Fs，无法跨采样率输出恒定 12.288MHz。但 SAI 内核固定 49.152MHz、HAL 自动调 MCKDIV，使每个采样率下 **MCLK = 256×Fs**（48k→12.288M、16k→4.096M、8k→2.048M）。因此 ES8311 看到的 **MCLK/Fs 比恒为 256**，时钟系数对三档**完全相同**，只需一组。

系数取自 ES8311 官方 coeff_div 表 ratio=256 行（ESP-ADF 参考驱动）：
pre_div=1, pre_multi=1, adc_div=1, dac_div=1, fs_mode=0, adc_osr=0x10, bclk_div=4, lrck=256(0x00/0xFF)。
按 ES8311 寄存器位打包得：

```c
/* MCLK/Fs = 256（恒定）→ 三档采样率共用一组时钟系数。 */
#define ES_REG02_CLK   0x00   /* pre_div=1,pre_multi=1 → 低3位保留,高位0 */
#define ES_REG03_CLK   0x10   /* fs_mode=0<<6 | adc_osr=0x10 */
#define ES_REG04_CLK   0x10   /* dac_osr(adc-only 无害) */
#define ES_REG05_CLK   0x00   /* (adc_div-1)<<4 | (dac_div-1) = 0 */
#define ES_REG06_CLK   0x03   /* bclk_div=4 (<19 → 写 bclk_div-1=3),高3位保留 */
#define ES_REG07_CLK   0x00   /* lrck_h */
#define ES_REG08_CLK   0xFF   /* lrck_l (lrck=0x00FF=255 → /256) */
```

> 这些值对三档采样率通用。台架（Task 7）用 1kHz 已知音验证音高正确即证明时钟链对。

- [ ] **Step 4: es8311.c —— ADC 初始化序列**

```c
int ES8311_InitAdc(uint32_t sample_rate_hz, uint16_t gain_db)
{
  (void)sample_rate_hz;            /* MCLK/Fs 恒为 256，时钟系数与采样率无关 */
  /* 复位 */
  if (es_write(0x00, 0x1F) != 0) return -1;
  osDelay(1);
  es_write(0x45, 0x00);
  /* 时钟管理：MCLK 来源、使能 ADC 时钟 */
  es_write(0x01, 0x30);            /* mclk on, adc clk on */
  es_write(0x02, ES_REG02_CLK);
  es_write(0x03, ES_REG03_CLK);
  es_write(0x16, 0x24);
  es_write(0x04, ES_REG04_CLK);
  es_write(0x05, ES_REG05_CLK);
  es_write(0x06, ES_REG06_CLK);
  es_write(0x07, ES_REG07_CLK);
  es_write(0x08, ES_REG08_CLK);
  /* SDP out（ADC）格式：I2S, 16-bit */
  es_write(0x09, 0x00);
  es_write(0x0A, 0x00);            /* 0x00 = I2S 16-bit；左/右对齐改此寄存器 */
  /* 系统/模拟上电 */
  es_write(0x0D, 0x01);
  es_write(0x0E, 0x02);            /* 模拟 mic PGA 上电 */
  es_write(0x12, 0x00);
  es_write(0x13, 0x10);
  es_write(0x10, 0x03);            /* vmid */
  es_write(0x11, 0x7B);
  es_write(0x00, 0x80);            /* 从机模式、整体上电 */
  /* ADC 数字：选择模拟 mic（MIC1P/N 差分），关闭 DAC 回环 */
  es_write(0x44, 0x08);            /* adc 数据来源 = adc（非回环） */
  es_write(0x17, 0xBF);            /* ADC 数字音量（0dB 附近） */
  /* mic PGA 增益：0x14 低 4 位选增益档（约 0..42dB，3dB/档）+ 选 MIC1 通道 */
  uint8_t pga = (uint8_t)((gain_db / 3U) & 0x0FU);
  es_write(0x14, (uint8_t)(0x10 | pga));  /* 0x10=选模拟差分 mic1，低4位增益 */
  es_write(0x15, 0x00);
  es_write(0x1B, 0x0A);
  es_write(0x1C, 0x6A);            /* ADC 高通/格式 */
  return 0;
}

void ES8311_PowerDown(void)
{
  es_write(0x0E, 0xFF);   /* 模拟下电 */
  es_write(0x12, 0x02);
  es_write(0x00, 0x00);   /* 复位/下电 */
}
```

```c
#include <string.h>

/* DMA 目标：双半缓冲，半字（int16）。每半 1024 样本 = 2KB。 */
#define MIC_DMA_HALF_SAMPLES   1024U
static int16_t s_dma_buf[MIC_DMA_HALF_SAMPLES * 2U];
static volatile uint32_t s_dropped = 0U;

extern uint32_t AppRing_WriteMic(const uint8_t *src, uint32_t len);

static void mic_push(const int16_t *p, uint32_t nsamp)
{
  uint32_t w = AppRing_WriteMic((const uint8_t *)p, nsamp * 2U);
  if (w == 0U) s_dropped += nsamp * 2U;
}

void HAL_SAI_RxHalfCpltCallback(SAI_HandleTypeDef *hsai)
{
  (void)hsai;
  mic_push(&s_dma_buf[0], MIC_DMA_HALF_SAMPLES);
}
void HAL_SAI_RxCpltCallback(SAI_HandleTypeDef *hsai)
{
  (void)hsai;
  mic_push(&s_dma_buf[MIC_DMA_HALF_SAMPLES], MIC_DMA_HALF_SAMPLES);
}

int Mic_Start(uint32_t sample_rate_hz, uint16_t gain_db)
{
  PaEn_Set(1);
  osDelay(50);                              /* 等 codec 供电稳定 */
  if (ES8311_Probe() != 0) { PaEn_Set(0); return -1; }
  if (ES8311_InitAdc(sample_rate_hz, gain_db) != 0) { PaEn_Set(0); return -2; }
  if (MX_SAI1_Init(sample_rate_hz) != HAL_OK) { ES8311_PowerDown(); PaEn_Set(0); return -3; }
  s_dropped = 0U;
  if (HAL_SAI_Receive_DMA(&hsai_BlockA1, (uint8_t *)s_dma_buf,
                          MIC_DMA_HALF_SAMPLES * 2U) != HAL_OK) {  /* 总样本数=两半 */
    MX_SAI1_DeInit(); ES8311_PowerDown(); PaEn_Set(0); return -4;
  }
  return 0;
}

void Mic_Stop(void)
{
  HAL_SAI_DMAStop(&hsai_BlockA1);
  MX_SAI1_DeInit();
  ES8311_PowerDown();
  PaEn_Set(0);
}

uint32_t Mic_GetDropped(void) { return s_dropped; }
```

## 自检（计划 vs 设计）

- **Spec 覆盖**：codec 驱动(T1) / SAI+DMA(T0,T5) / ring(T5,T6) / WAV(T4) / 配置(T2) / JSON 持久化(T3) / 电源门控(T0) / I2C2 共享(T1 加锁说明) / SD+联动(T6) / 验证(T7) —— 设计各点均有对应任务。USB 串流明确划入 Plan 2。
- **占位扫描**：无 "TODO/TBD"。两处需"对照官方资料确认"的是 `kClk` 分频系数(T1)与 PLL/GPIO 脚位(T0)——均给了具体起始值 + 台架校验步骤，属硬件标定而非占位。
- **类型一致**：`AcqMicCfg_t`/`g_ring_mic`/`Mic_Start`/`ES8311_InitAdc`/`FatFs_SD_FILE_MIC_WAV`/`FatFs_SD_WavCreate`/`FatFs_SD_WavFinalize`/`AppRing_WriteMic` 在各任务间命名一致。

## 已知风险与依赖

1. **SAI 时钟树**：让 SAI1 内核时钟成为 12.288MHz 整数倍需占用一个 PLL（PLL2/PLL3）。实现前确认该 PLL 未被其它外设占用（SDMMC/USB 用的是各自时钟源）。
2. **`kClk` 系数**：12 页数据手册无完整寄存器表，分频系数来自 ES8311 user guide / ESP-ADF 参考驱动；台架 1kHz 测试是最终判据。
3. **GPDMA 通道**：选一个未被 SDMMC/SPI 占用的 GPDMA 通道；中断优先级低于 SDMMC(6)。
4. **I2C2 共享**：codec 初始化（仅 start 时一次）与 RTC `AppTime_Sync` 可能并发 → 在两处 I2C2 调用外加同一把互斥锁。
